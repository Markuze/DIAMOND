% Styles shared by the evaluation's measured figures (rx.tex, tx.tex, fwd.tex).
% Receive/forwarding use categorical axes; send uses the buffer/range-size style below.
% \input once, before the first figure. No unmeasured core-count panels are shown.
%
% One visual language in all three, so a reader learns it once:
%   black, solid        memory the kernel lends: the named mode (circles) and the mapped mode (squares)
%   gray                the existing zero-copy mechanisms: application memory lent to the kernel
%                       (dashed: pinned per operation; solid: registered), and payload the kernel
%                       keeps for files or pipes (dotted: sendfile, splice)
%   black, dash-dotted  copying
% Identity is never carried by the gray level alone: every series has its own mark and dash.
% Marks are drawn solid whatever the line's dash, or a dashed line breaks its own marks.
\pgfplotsset{
  bysize/.style={
    width=0.49\textwidth, height=3.9cm,
    xmode=log, ymode=log, log basis x=2, log ticks with fixed point,
    xmin=0.7, xmax=1450,
    xtick={1,4,16,64,256,1024}, xticklabels={1,4,16,64,256,1024},
    yticklabel style={/pgf/number format/fixed, /pgf/number format/precision=2},
    xlabel={message size (KiB)},
    grid=major, grid style={gray!20},
    tick label style={font=\scriptsize}, label style={font=\footnotesize},
    title style={font=\footnotesize, yshift=-0.5ex},
    unbounded coords=discard,
    every axis plot/.append style={line width=0.7pt, mark size=1.7pt},
    legend cell align=left,
    legend style={font=\scriptsize, draw=none, fill=none, column sep=0.5em},
  },
  % One standard deviation over the cell's runs, from the <column>_sd column.
  sdbars/.style={error bars/y dir=both, error bars/y explicit,
    error bars/error bar style={line width=0.35pt, solid},
    error bars/error mark options={rotate=90, mark size=1.1pt, line width=0.35pt, solid}},
  % Copying.
  mcopy/.style={black, densely dashdotted, mark=triangle*, mark options={solid}},
  % Application memory lent to the kernel.
  mmsgzc/.style={black!55, dashed, mark=square, mark options={solid}},
  mzcpin/.style={black!55, dashed, mark=o, mark options={solid}},
  mzcreg/.style={black!55, solid, mark=diamond},
  mzcvec/.style={black!55, solid, line width=1.1pt, mark=diamond*},
  % Payload the kernel keeps, with a fixed fate.
  mfixed/.style={black!55, densely dotted, mark=x, mark size=2.4pt, mark options={solid, line width=0.8pt}},
  % Memory the kernel lends.
  mnamed/.style={black, solid, mark=*},
  mnamedvec/.style={black, solid, line width=1.1pt, mark=pentagon*},
  mdiscard/.style={black, solid, mark=o},
  mbound/.style={black, solid, mark=diamond*, mark size=2.1pt},
  mmapped/.style={black, solid, mark=square*},
  mmappedzc/.style={black, dashed, mark=square, mark options={solid}},
}
